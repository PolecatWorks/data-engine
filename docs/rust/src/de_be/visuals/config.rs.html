1//! Modular configuration for Data Visuals & Semantic Relationship Topology (Specs 13 & 14).
2
3use serde::{Deserialize, Serialize};
4use thiserror::Error;
5use url::Url;
6
7#[derive(Debug, Error, PartialEq)]
8pub enum VisualsConfigError {
9    #[error("visuals.max_chart_points must be between 10 and 100000, got {0}")]
10    InvalidMaxChartPoints(usize),
11    #[error("visuals.max_topology_nodes must be between 10 and 5000, got {0}")]
12    InvalidMaxTopologyNodes(usize),
13    #[error("visuals.max_topology_edges must be between 10 and 20000, got {0}")]
14    InvalidMaxTopologyEdges(usize),
15    #[error("visuals.default_ego_hops must be between 1 and 5, got {0}")]
16    InvalidDefaultEgoHops(u8),
17    #[error("visuals.graphrag_base_url must use http:// or https:// scheme when set")]
18    InvalidGraphragBaseUrl,
19}
20
21#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
22pub struct VisualsConfig {
23    pub max_chart_points: usize,
24    pub max_topology_nodes: usize,
25    pub max_topology_edges: usize,
26    pub default_ego_hops: u8,
27    pub graphrag_base_url: Option<Url>,
28}
29
30impl Default for VisualsConfig {
31    fn default() -> Self {
32        Self {
33            max_chart_points: 5000,
34            max_topology_nodes: 250,
35            max_topology_edges: 500,
36            default_ego_hops: 2,
37            graphrag_base_url: None,
38        }
39    }
40}
41
42impl VisualsConfig {
43    pub fn validate(&self) -> Result<(), VisualsConfigError> {
44        if self.max_chart_points < 10 || self.max_chart_points > 100_000 {
45            return Err(VisualsConfigError::InvalidMaxChartPoints(
46                self.max_chart_points,
47            ));
48        }
49        if self.max_topology_nodes < 10 || self.max_topology_nodes > 5_000 {
50            return Err(VisualsConfigError::InvalidMaxTopologyNodes(
51                self.max_topology_nodes,
52            ));
53        }
54        if self.max_topology_edges < 10 || self.max_topology_edges > 20_000 {
55            return Err(VisualsConfigError::InvalidMaxTopologyEdges(
56                self.max_topology_edges,
57            ));
58        }
59        if self.default_ego_hops == 0 || self.default_ego_hops > 5 {
60            return Err(VisualsConfigError::InvalidDefaultEgoHops(
61                self.default_ego_hops,
62            ));
63        }
64        if let Some(ref url) = self.graphrag_base_url {
65            let scheme = url.scheme();
66            if scheme != "http" && scheme != "https" {
67                return Err(VisualsConfigError::InvalidGraphragBaseUrl);
68            }
69        }
70        Ok(())
71    }
72}

74#[cfg(test)]
75mod tests {
76    use super::*;
77
78    #[test]
79    fn test_visuals_config_validation_rejects_out_of_range_bounds() {
80        assert!(VisualsConfig::default().validate().is_ok());
81
82        let with_http_url = VisualsConfig {
83            graphrag_base_url: Some(Url::parse("http://graphrag:8080/api/graphrag/v1").unwrap()),
84            ..Default::default()
85        };
86        assert!(with_http_url.validate().is_ok());
87
88        let too_few_points = VisualsConfig {
89            max_chart_points: 5,
90            ..Default::default()
91        };
92        assert_eq!(
93            too_few_points.validate(),
94            Err(VisualsConfigError::InvalidMaxChartPoints(5))
95        );
96
97        let too_few_nodes = VisualsConfig {
98            max_topology_nodes: 3,
99            ..Default::default()
100        };
101        assert_eq!(
102            too_few_nodes.validate(),
103            Err(VisualsConfigError::InvalidMaxTopologyNodes(3))
104        );
105
106        let zero_hops = VisualsConfig {
107            default_ego_hops: 0,
108            ..Default::default()
109        };
110        assert_eq!(
111            zero_hops.validate(),
112            Err(VisualsConfigError::InvalidDefaultEgoHops(0))
113        );
114
115        let excessive_hops = VisualsConfig {
116            default_ego_hops: 6,
117            ..Default::default()
118        };
119        assert_eq!(
120            excessive_hops.validate(),
121            Err(VisualsConfigError::InvalidDefaultEgoHops(6))
122        );
123
124        let bad_url = VisualsConfig {
125            graphrag_base_url: Some(Url::parse("ftp://graphrag:8080").unwrap()),
126            ..Default::default()
127        };
128        assert_eq!(
129            bad_url.validate(),
130            Err(VisualsConfigError::InvalidGraphragBaseUrl)
131        );
132    }
