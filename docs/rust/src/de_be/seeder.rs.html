52/// Simple deterministic SplitMix64 RNG so our test suite has zero platform-specific RNG variance.
53struct DeterministicRng {
54    state: u64,
55}
56
57impl DeterministicRng {
58    fn new(seed: u64) -> Self {
59        Self { state: seed }
60    }
61
62    fn next_u64(&mut self) -> u64 {
63        self.state = self.state.wrapping_add(0x9E3779B97F4A7C15);
64        let mut z = self.state;
65        z = (z ^ (z >> 30)).wrapping_mul(0xBF58476D1CE4E5B9);
66        z = (z ^ (z >> 27)).wrapping_mul(0x94D049BB133111EB);
67        z ^ (z >> 31)
68    }
69
70    fn next_usize(&mut self, bound: usize) -> usize {
71        (self.next_u64() as usize) % bound
72    }
73}

75fn write_parquet_batch(path: &Path, batch: &RecordBatch) -> Result<(), SeederError> {
76    let file = File::create(path)?;
77    let mut writer = ArrowWriter::try_new(file, batch.schema(), None)?;
78    writer.write(batch)?;
79    writer.close()?;
80    Ok(())
81}

83pub fn seed_validation_suite(
84    seed: u64,
85    base_row_count: usize,
86    output_dir: &Path,
87) -> Result<GroundTruthManifest, SeederError> {
88    fs::create_dir_all(output_dir)?;
89    let mut rng = DeterministicRng::new(seed);
90
91    let regions = ["EMEA", "AMER", "APAC", "LATAM"];
92    let tiers = ["Enterprise", "MidMarket", "Growth"];
93    let statuses = ["completed", "completed", "completed", "shipped"];
94
95    // 1. crm_customers.parquet
96    let mut party_uuids = Vec::with_capacity(base_row_count);
97    let mut customer_names = Vec::with_capacity(base_row_count);
98    let mut customer_regions = Vec::with_capacity(base_row_count);
99    let mut customer_tiers = Vec::with_capacity(base_row_count);
100
101    for idx in 0..base_row_count {
102        let uid = format!("PTY-{:04X}-{:06}", (seed & 0xFFFF), idx + 1);
103        party_uuids.push(uid);
104        customer_names.push(format!("Acme Corp {:04}", idx + 1));
105        customer_regions.push(regions[rng.next_usize(regions.len())].to_string());
106        customer_tiers.push(tiers[rng.next_usize(tiers.len())].to_string());
107    }
108
109    let crm_schema = Arc::new(Schema::new(vec![
110        Field::new("party_uuid", DataType::Utf8, false),
111        Field::new("customer_name", DataType::Utf8, false),
112        Field::new("region", DataType::Utf8, false),
113        Field::new("tier", DataType::Utf8, false),
114    ]));
115    let crm_batch = RecordBatch::try_new(
116        crm_schema,
117        vec![
118            Arc::new(StringArray::from(party_uuids.clone())),
119            Arc::new(StringArray::from(customer_names)),
120            Arc::new(StringArray::from(customer_regions)),
121            Arc::new(StringArray::from(customer_tiers)),
122        ],
123    )?;
124    write_parquet_batch(&output_dir.join("crm_customers.parquet"), &crm_batch)?;
125
126    // 2. sales_orders.parquet (TRAP-02: cust_ref_key -> crm_customers.party_uuid with >= 97% overlap)
127    // Also drives TRAP-03: order_items (2..3 rows/order) x order_payments (2..3 installments/order)
128    let order_count = base_row_count * 2;
129    let mut order_ids = Vec::with_capacity(order_count);
130    let mut cust_ref_keys = Vec::with_capacity(order_count);
131    let mut order_totals = Vec::with_capacity(order_count);
132    let mut order_statuses = Vec::with_capacity(order_count);
133
134    let mut item_ids = Vec::new();
135    let mut item_order_ids = Vec::new();
136    let mut item_skus = Vec::new();
137    let mut item_prices = Vec::new();
138
139    let mut payment_ids = Vec::new();
140    let mut payment_order_ids = Vec::new();
141    let mut payment_seqs = Vec::new();
142    let mut payment_amounts = Vec::new();
143
144    let mut true_total_order_revenue_usd = 0.0_f64;
145    let mut naive_fanout_inflated_revenue_usd = 0.0_f64;
146
147    for ord_idx in 0..order_count {
148        let oid = format!("ORD-{:06}", ord_idx + 1);
149        // Ensure every customer is referenced and ~98.5% of orders map to known party_uuids
150        let cust_key = if ord_idx < base_row_count {
151            party_uuids[ord_idx].clone()
152        } else if ord_idx % 60 == 0 {
153            format!("PTY-GHOST-{:06}", ord_idx)
154        } else {
155            party_uuids[rng.next_usize(base_row_count)].clone()
156        };
157
158        // 2 or 3 line items per order
159        let num_items = if ord_idx % 2 == 0 { 2 } else { 3 };
160        let mut this_order_total = 0.0_f64;
161        for i in 0..num_items {
162            let price = 40.0 + ((rng.next_usize(160) as f64) * 1.5);
163            this_order_total += price;
164            item_ids.push(format!("ITM-{:06}-{}", ord_idx + 1, i + 1));
165            item_order_ids.push(oid.clone());
166            item_skus.push(format!("SKU-{:03}", (i + ord_idx) % 50 + 1));
167            item_prices.push(price);
168        }
169
170        // 2 payment installments per order summing to this_order_total
171        let num_payments = 2_usize;
172        let installment_amt = this_order_total / (num_payments as f64);
173        for p in 0..num_payments {
174            payment_ids.push(format!("PAY-{:06}-{}", ord_idx + 1, p + 1));
175            payment_order_ids.push(oid.clone());
176            payment_seqs.push((p + 1) as i64);
177            payment_amounts.push(installment_amt);
178        }
179
180        // When order_items (num_items rows) is joined directly to order_payments (num_payments rows)
181        // on order_id without pre-aggregation, each payment row is repeated num_items times!
182        true_total_order_revenue_usd += this_order_total;
183        naive_fanout_inflated_revenue_usd += this_order_total * (num_items as f64);
184
185        order_ids.push(oid);
186        cust_ref_keys.push(cust_key);
187        order_totals.push(this_order_total);
188        order_statuses.push(statuses[ord_idx % statuses.len()].to_string());
189    }
190
191    let orders_schema = Arc::new(Schema::new(vec![
192        Field::new("order_id", DataType::Utf8, false),
193        Field::new("cust_ref_key", DataType::Utf8, false),
194        Field::new("order_total_usd", DataType::Float64, false),
195        Field::new("status", DataType::Utf8, false),
196    ]));
197    let orders_batch = RecordBatch::try_new(
198        orders_schema,
199        vec![
200            Arc::new(StringArray::from(order_ids)),
201            Arc::new(StringArray::from(cust_ref_keys)),
202            Arc::new(Float64Array::from(order_totals)),
203            Arc::new(StringArray::from(order_statuses)),
204        ],
205    )?;
206    write_parquet_batch(&output_dir.join("sales_orders.parquet"), &orders_batch)?;
207
208    // 3. order_items.parquet
209    let items_schema = Arc::new(Schema::new(vec![
210        Field::new("item_id", DataType::Utf8, false),
211        Field::new("order_id", DataType::Utf8, false),
212        Field::new("sku", DataType::Utf8, false),
213        Field::new("item_price_usd", DataType::Float64, false),
214    ]));
215    let items_batch = RecordBatch::try_new(
216        items_schema,
217        vec![
218            Arc::new(StringArray::from(item_ids)),
219            Arc::new(StringArray::from(item_order_ids)),
220            Arc::new(StringArray::from(item_skus)),
221            Arc::new(Float64Array::from(item_prices)),
222        ],
223    )?;
224    write_parquet_batch(&output_dir.join("order_items.parquet"), &items_batch)?;
225
226    // 4. order_payments.parquet
227    let payments_schema = Arc::new(Schema::new(vec![
228        Field::new("payment_id", DataType::Utf8, false),
229        Field::new("order_id", DataType::Utf8, false),
230        Field::new("installment_seq", DataType::Int64, false),
231        Field::new("payment_amount_usd", DataType::Float64, false),
232    ]));
233    let payments_batch = RecordBatch::try_new(
234        payments_schema,
235        vec![
236            Arc::new(StringArray::from(payment_ids)),
237            Arc::new(StringArray::from(payment_order_ids)),
238            Arc::new(Int64Array::from(payment_seqs)),
239            Arc::new(Float64Array::from(payment_amounts)),
240        ],
241    )?;
242    write_parquet_batch(&output_dir.join("order_payments.parquet"), &payments_batch)?;
243
244    // 5 & 6. TRAP-01: billing_subscriptions.parquet vs marketing_leads.parquet
245    // Both have sequential surrogate `account_id` (1..=base_row_count), creating a 100% value overlap
246    // on an auto-increment integer column that actually represents two completely disjoint namespaces!
247    let mut billing_account_ids = Vec::with_capacity(base_row_count);
248    let mut billing_tax_ids = Vec::with_capacity(base_row_count);
249    let mut billing_mrr = Vec::with_capacity(base_row_count);
250
251    let mut marketing_account_ids = Vec::with_capacity(base_row_count);
252    let mut marketing_campaigns = Vec::with_capacity(base_row_count);
253    let mut marketing_click_ids = Vec::with_capacity(base_row_count);
254
255    for idx in 0..base_row_count {
256        let surrogate_id = (idx + 1) as i64;
257        billing_account_ids.push(surrogate_id);
258        billing_tax_ids.push(format!("TAX-ENT-{:05}", idx + 1));
259        billing_mrr.push(500.0 + (idx as f64) * 15.0);
260
261        marketing_account_ids.push(surrogate_id);
262        marketing_campaigns.push(format!("CMP-MKT-{:05}", idx + 1));
263        marketing_click_ids.push(format!("CLK-AD-{:06}", idx + 1));
264    }
265
266    let billing_schema = Arc::new(Schema::new(vec![
267        Field::new("account_id", DataType::Int64, false),
268        Field::new("billing_tax_id", DataType::Utf8, false),
269        Field::new("mrr_usd", DataType::Float64, false),
270    ]));
271    let billing_batch = RecordBatch::try_new(
272        billing_schema,
273        vec![
274            Arc::new(Int64Array::from(billing_account_ids)),
275            Arc::new(StringArray::from(billing_tax_ids)),
276            Arc::new(Float64Array::from(billing_mrr)),
277        ],
278    )?;
279    write_parquet_batch(
280        &output_dir.join("billing_subscriptions.parquet"),
281        &billing_batch,
282    )?;
283
284    let marketing_schema = Arc::new(Schema::new(vec![
285        Field::new("account_id", DataType::Int64, false),
286        Field::new("lead_campaign_code", DataType::Utf8, false),
287        Field::new("ad_click_id", DataType::Utf8, false),
288    ]));
289    let marketing_batch = RecordBatch::try_new(
290        marketing_schema,
291        vec![
292            Arc::new(Int64Array::from(marketing_account_ids)),
293            Arc::new(StringArray::from(marketing_campaigns)),
294            Arc::new(StringArray::from(marketing_click_ids)),
295        ],
296    )?;
297    write_parquet_batch(
298        &output_dir.join("marketing_leads.parquet"),
299        &marketing_batch,
300    )?;
301
302    // 7 & 8. TRAP-04: Temporal Schema & 100x Unit Drift (payment_events_wave1 vs payment_events_wave3)
303    let mut w1_event_ids = Vec::with_capacity(base_row_count);
304    let mut w1_user_ids = Vec::with_capacity(base_row_count);
305    let mut w1_amounts_usd = Vec::with_capacity(base_row_count);
306
307    let mut w3_event_ids = Vec::with_capacity(base_row_count);
308    let mut w3_client_ids = Vec::with_capacity(base_row_count);
309    let mut w3_amounts_cents = Vec::with_capacity(base_row_count);
310
311    for idx in 0..base_row_count {
312        let amt_usd = 100.0 + ((idx % 41) as f64); // Mean ~120.0 USD
313        w1_event_ids.push(format!("EVT-W1-{:05}", idx + 1));
314        w1_user_ids.push(party_uuids[idx].clone());
315        w1_amounts_usd.push(amt_usd);
316
317        w3_event_ids.push(format!("EVT-W3-{:05}", idx + 1));
318        w3_client_ids.push(party_uuids[idx].clone());
319        w3_amounts_cents.push(amt_usd * 100.0); // 100x shift to cents (~12000.0)
320    }
321
322    let wave1_mean: f64 = w1_amounts_usd.iter().sum::<f64>() / (base_row_count as f64);
323    let wave3_mean: f64 = w3_amounts_cents.iter().sum::<f64>() / (base_row_count as f64);
324
325    let w1_schema = Arc::new(Schema::new(vec![
326        Field::new("event_id", DataType::Utf8, false),
327        Field::new("user_id", DataType::Utf8, false),
328        Field::new("amount", DataType::Float64, false),
329    ]));
330    let w1_batch = RecordBatch::try_new(
331        w1_schema,
332        vec![
333            Arc::new(StringArray::from(w1_event_ids)),
334            Arc::new(StringArray::from(w1_user_ids)),
335            Arc::new(Float64Array::from(w1_amounts_usd)),
336        ],
337    )?;
338    write_parquet_batch(
339        &output_dir.join("payment_events_wave1.parquet"),
340        &w1_batch,
341    )?;
342
343    let w3_schema = Arc::new(Schema::new(vec![
344        Field::new("event_id", DataType::Utf8, false),
345        Field::new("client_id", DataType::Utf8, false),
346        Field::new("amount", DataType::Float64, false),
347    ]));
348    let w3_batch = RecordBatch::try_new(
349        w3_schema,
350        vec![
351            Arc::new(StringArray::from(w3_event_ids)),
352            Arc::new(StringArray::from(w3_client_ids)),
353            Arc::new(Float64Array::from(w3_amounts_cents)),
354        ],
355    )?;
356    write_parquet_batch(
357        &output_dir.join("payment_events_wave3.parquet"),
358        &w3_batch,
359    )?;
360
361    let expected_fanout_multiplier =
362        (naive_fanout_inflated_revenue_usd / true_total_order_revenue_usd * 100.0).round() / 100.0;
363
364    let manifest = GroundTruthManifest {
365        seed,
366        base_row_count,
367        generated_files: vec![
368            "crm_customers.parquet".to_string(),
369            "sales_orders.parquet".to_string(),
370            "order_items.parquet".to_string(),
371            "order_payments.parquet".to_string(),
372            "billing_subscriptions.parquet".to_string(),
373            "marketing_leads.parquet".to_string(),
374            "payment_events_wave1.parquet".to_string(),
375            "payment_events_wave3.parquet".to_string(),
376        ],
377        true_total_order_revenue_usd: (true_total_order_revenue_usd * 100.0).round() / 100.0,
378        naive_fanout_inflated_revenue_usd: (naive_fanout_inflated_revenue_usd * 100.0).round()
379            / 100.0,
380        expected_fanout_multiplier,
381        wave1_payment_mean_usd: (wave1_mean * 100.0).round() / 100.0,
382        wave3_payment_mean_cents: (wave3_mean * 100.0).round() / 100.0,
383        planted_traps: vec![
384            PlantedTrapSpec {
385                trap_id: "TRAP-01".to_string(),
386                category: "homonym_chasm".to_string(),
387                left_dataset: "billing_subscriptions".to_string(),
388                left_field: "account_id".to_string(),
389                right_dataset: "marketing_leads".to_string(),
390                right_field: "account_id".to_string(),
391                expected_action: "require_steward_verification_then_forbidden_join".to_string(),
392                description: "Both tables use sequential integer surrogate account_id (1..N) for unrelated entity domains (Billing Tax Accounts vs Ad Marketing Leads).".to_string(),
393            },
394            PlantedTrapSpec {
395                trap_id: "TRAP-02".to_string(),
396                category: "cryptic_synonym_fk".to_string(),
397                left_dataset: "sales_orders".to_string(),
398                left_field: "cust_ref_key".to_string(),
399                right_dataset: "crm_customers".to_string(),
400                right_field: "party_uuid".to_string(),
401                expected_action: "sync_joinable_with".to_string(),
402                description: "Zero column-name similarity, but >98% value inclusion ratio and clean N:1 cardinality.".to_string(),
403            },
404            PlantedTrapSpec {
405                trap_id: "TRAP-03".to_string(),
406                category: "nm_fanout_inflation".to_string(),
407                left_dataset: "order_items".to_string(),
408                left_field: "order_id".to_string(),
409                right_dataset: "order_payments".to_string(),
410                right_field: "order_id".to_string(),
411                expected_action: "pre_aggregate_before_join".to_string(),
412                description: "Joining multi-row order_items to multi-row order_payments on order_id inflates payment revenue by ~2.5x.".to_string(),
413            },
414            PlantedTrapSpec {
415                trap_id: "TRAP-04".to_string(),
416                category: "temporal_schema_and_unit_drift".to_string(),
417                left_dataset: "payment_events_wave1".to_string(),
418                left_field: "amount".to_string(),
419                right_dataset: "payment_events_wave3".to_string(),
420                right_field: "amount".to_string(),
421                expected_action: "mark_drifted_and_require_unit_conversion".to_string(),
422                description: "Wave 3 renames user_id -> client_id and shifts amount from USD (~120.0) to integer cents (~12000.0, 100x scale shift).".to_string(),
423            },
424        ],
425    };
426
427    let manifest_path: PathBuf = output_dir.join("ground_truth_manifest.json");
428    let manifest_json = serde_json::to_string_pretty(&manifest)?;
429    fs::write(manifest_path, manifest_json)?;
430
431    Ok(manifest)
432}

435mod tests {
436    use super::*;
437    use tempfile::tempdir;

439    #[test]
440    fn test_seed_validation_suite_creates_all_parquet_files_and_manifest() {
441        let dir = tempdir().expect("create temp dir");
442        let manifest =
443            seed_validation_suite(42, 100, dir.path()).expect("seed validation suite succeeds");
444
445        assert_eq!(manifest.generated_files.len(), 8);
446        for file_name in &manifest.generated_files {
447            let p = dir.path().join(file_name);
448            assert!(p.exists(), "expected generated file {:?} to exist", p);
449            let meta = fs::metadata(&p).expect("file metadata");
450            assert!(meta.len() > 100, "expected non-empty parquet file {:?}", p);
451        }
452        assert!(dir.path().join("ground_truth_manifest.json").exists());
453        assert!(manifest.expected_fanout_multiplier >= 2.4);
454        assert_eq!(
455            (manifest.wave3_payment_mean_cents / manifest.wave1_payment_mean_usd).round(),
456            100.0
457        );
458    }
459
460    #[test]
461    fn test_seed_is_strictly_deterministic() {
462        let dir1 = tempdir().expect("dir1");
463        let dir2 = tempdir().expect("dir2");
464
465        let m1 = seed_validation_suite(42, 120, dir1.path()).expect("m1");
466        let m2 = seed_validation_suite(42, 120, dir2.path()).expect("m2");
467
468        assert_eq!(m1, m2);
469    }
