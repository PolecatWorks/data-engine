55/// Simple deterministic SplitMix64 RNG so our test suite has zero platform-specific RNG variance.
56struct DeterministicRng {
57    state: u64,
58}
59
60impl DeterministicRng {
61    fn new(seed: u64) -> Self {
62        Self { state: seed }
63    }
64
65    fn next_u64(&mut self) -> u64 {
66        self.state = self.state.wrapping_add(0x9E3779B97F4A7C15);
67        let mut z = self.state;
68        z = (z ^ (z >> 30)).wrapping_mul(0xBF58476D1CE4E5B9);
69        z = (z ^ (z >> 27)).wrapping_mul(0x94D049BB133111EB);
70        z ^ (z >> 31)
71    }
72
73    fn next_usize(&mut self, bound: usize) -> usize {
74        (self.next_u64() as usize) % bound
75    }
76
77    fn next_f64(&mut self) -> f64 {
78        // Uniform in (0, 1) strictly avoiding 0.0 for Box-Muller log
79        let bits = (self.next_u64() >> 11) as f64;
80        (bits + 1.0) / ((1u64 << 53) as f64 + 1.0)
81    }
82
83    fn next_standard_normal(&mut self) -> f64 {
84        let u1 = self.next_f64();
85        let u2 = self.next_f64();
86        (-2.0 * u1.ln()).sqrt() * (2.0 * std::f64::consts::PI * u2).cos()
87    }
88}

90fn write_parquet_batch(path: &Path, batch: &RecordBatch) -> Result<(), SeederError> {
91    let file = File::create(path)?;
92    let mut writer = ArrowWriter::try_new(file, batch.schema(), None)?;
93    writer.write(batch)?;
94    writer.close()?;
95    Ok(())
96}

477fn sample_mean_and_var(xs: &[f64]) -> (f64, f64) {
478    let n = xs.len() as f64;
479    let mean = xs.iter().sum::<f64>() / n;
480    let var = xs.iter().map(|v| (v - mean).powi(2)).sum::<f64>() / (n - 1.0).max(1.0);
481    (mean, var)
482}
483
484fn pearson_r(a: &[f64], b: &[f64]) -> f64 {
485    let (ma, _) = sample_mean_and_var(a);
486    let (mb, _) = sample_mean_and_var(b);
487    let mut num = 0.0;
488    let mut da = 0.0;
489    let mut db = 0.0;
490    for (x, y) in a.iter().zip(b.iter()) {
491        let dx = x - ma;
492        let dy = y - mb;
493        num += dx * dy;
494        da += dx * dx;
495        db += dy * dy;
496    }
497    if da <= 0.0 || db <= 0.0 {
498        0.0
499    } else {
500        num / (da.sqrt() * db.sqrt())
501    }
502}

504/// Generate the 4 deterministic Statistical Chaos Trap datasets (`STAT-TRAP-01..04`) per Spec 12.
505pub fn seed_statistical_chaos_traps(
506    seed: u64,
507    output_dir: &Path,
508) -> Result<Vec<PlantedTrapSpec>, SeederError> {
509    fs::create_dir_all(output_dir)?;
510    let mut rng = DeterministicRng::new(seed ^ 0x57A7_78A9_C0DE_u64);
511
512    // -------------------------------------------------------------------------
513    // STAT-TRAP-01: ab_simpsons_paradox.parquet
514    // Enterprise: Control 80% (160/200) vs Treatment 78% (624/800)  [-2pp]
515    // SMB:        Control 40% (320/800) vs Treatment 38% (76/200)   [-2pp]
516    // Aggregate:  Control 48% (480/1000) vs Treatment 70% (700/1000) [+22pp reversal!]
517    // -------------------------------------------------------------------------
518    let mut sp_user_ids = Vec::with_capacity(2000);
519    let mut sp_variants = Vec::with_capacity(2000);
520    let mut sp_segments = Vec::with_capacity(2000);
521    let mut sp_converted = Vec::with_capacity(2000);
522    let mut uid_seq = 1usize;
523
524    let cells = [
525        ("control", "Enterprise", 200usize, 160usize),
526        ("treatment", "Enterprise", 800usize, 624usize),
527        ("control", "SMB", 800usize, 320usize),
528        ("treatment", "SMB", 200usize, 76usize),
529    ];
530    for (variant, segment, total, positives) in cells {
531        for i in 0..total {
532            sp_user_ids.push(format!("USR-SP-{:05}", uid_seq));
533            sp_variants.push(variant.to_string());
534            sp_segments.push(segment.to_string());
535            sp_converted.push(if i < positives { 1.0 } else { 0.0 });
536            uid_seq += 1;
537        }
538    }
539    let sp_schema = Arc::new(Schema::new(vec![
540        Field::new("user_id", DataType::Utf8, false),
541        Field::new("variant", DataType::Utf8, false),
542        Field::new("segment", DataType::Utf8, false),
543        Field::new("converted", DataType::Float64, false),
544    ]));
545    let sp_batch = RecordBatch::try_new(
546        sp_schema,
547        vec![
548            Arc::new(StringArray::from(sp_user_ids)),
549            Arc::new(StringArray::from(sp_variants)),
550            Arc::new(StringArray::from(sp_segments)),
551            Arc::new(Float64Array::from(sp_converted)),
552        ],
553    )?;
554    write_parquet_batch(&output_dir.join("ab_simpsons_paradox.parquet"), &sp_batch)?;
555
556    // -------------------------------------------------------------------------
557    // STAT-TRAP-02: orders_heavy_tail.parquet
558    // Control (200): median ~100 + 2 extreme $50,000 whale outliers (naive mean ~599)
559    // Treatment (200): median ~118 with no whale outliers (naive mean ~118)
560    // Mann-Whitney U catches true median lift (p < 0.01) after normality fails.
561    // -------------------------------------------------------------------------
562    let mut ht_order_ids = Vec::with_capacity(400);
563    let mut ht_variants = Vec::with_capacity(400);
564    let mut ht_totals = Vec::with_capacity(400);
565
566    for i in 0..200usize {
567        ht_order_ids.push(format!("ORD-HT-C-{:04}", i + 1));
568        ht_variants.push("control".to_string());
569        let val = if i >= 198 {
570            50_000.0
571        } else {
572            (100.0 + rng.next_standard_normal() * 8.0).max(20.0)
573        };
574        ht_totals.push(val);
575    }
576    for i in 0..200usize {
577        ht_order_ids.push(format!("ORD-HT-T-{:04}", i + 1));
578        ht_variants.push("treatment".to_string());
579        let val = (118.0 + rng.next_standard_normal() * 8.0).max(20.0);
580        ht_totals.push(val);
581    }
582    let ht_schema = Arc::new(Schema::new(vec![
583        Field::new("order_id", DataType::Utf8, false),
584        Field::new("variant", DataType::Utf8, false),
585        Field::new("order_total_usd", DataType::Float64, false),
586    ]));
587    let ht_batch = RecordBatch::try_new(
588        ht_schema,
589        vec![
590            Arc::new(StringArray::from(ht_order_ids)),
591            Arc::new(StringArray::from(ht_variants)),
592            Arc::new(Float64Array::from(ht_totals)),
593        ],
594    )?;
595    write_parquet_batch(&output_dir.join("orders_heavy_tail.parquet"), &ht_batch)?;
596
597    // -------------------------------------------------------------------------
598    // STAT-TRAP-03: noise_segments_20.parquet
599    // 20 segments (SEG-01..SEG-20), 50 control + 50 treatment per segment.
600    // Guarantees >= 1 raw p < 0.05 false positive that vanishes to 0 under BH-FDR
601    // (adjusted_p_value > 0.25 for every segment).
602    // -------------------------------------------------------------------------
603    let mut ns_segments = Vec::with_capacity(2000);
604    let mut ns_variants = Vec::with_capacity(2000);
605    let mut ns_values = Vec::with_capacity(2000);
606
607    for seg_idx in 1..=20usize {
608        let seg_name = format!("SEG-{:02}", seg_idx);
609        let (ctrl, trt) = if seg_idx == 7 {
610            // Borderline noise spike: Welch |t| in [2.05, 2.30] -> raw p in [0.023, 0.043],
611            // so BH-FDR across 20 segments adjusts p to ~[0.46, 0.86] > 0.25.
612            let mut c = Vec::with_capacity(50);
613            let mut t = Vec::with_capacity(50);
614            for _ in 0..50 {
615                c.push(rng.next_standard_normal());
616                t.push(rng.next_standard_normal());
617            }
618            let (mc, vc) = sample_mean_and_var(&c);
619            let (mt, vt) = sample_mean_and_var(&t);
620            let se = (vc / 50.0 + vt / 50.0).sqrt();
621            let desired_diff = 2.15 * se;
622            let shift = desired_diff - (mt - mc);
623            for v in &mut t {
624                *v += shift;
625            }
626            (c, t)
627        } else {
628            // Pure Gaussian null; reject rare extreme draws (|t| >= 1.75) so no accidental
629            // ultra-low p-value survives BH-FDR adjustment.
630            loop {
631                let mut c = Vec::with_capacity(50);
632                let mut t = Vec::with_capacity(50);
633                for _ in 0..50 {
634                    c.push(rng.next_standard_normal());
635                    t.push(rng.next_standard_normal());
636                }
637                let (mc, vc) = sample_mean_and_var(&c);
638                let (mt, vt) = sample_mean_and_var(&t);
639                let se = (vc / 50.0 + vt / 50.0).sqrt();
640                let t_stat = ((mt - mc) / se).abs();
641                if t_stat < 1.75 {
642                    break (c, t);
643                }
644            }
645        };
646
647        for v in ctrl {
648            ns_segments.push(seg_name.clone());
649            ns_variants.push("control".to_string());
650            ns_values.push(v);
651        }
652        for v in trt {
653            ns_segments.push(seg_name.clone());
654            ns_variants.push("treatment".to_string());
655            ns_values.push(v);
656        }
657    }
658    let ns_schema = Arc::new(Schema::new(vec![
659        Field::new("segment_id", DataType::Utf8, false),
660        Field::new("variant", DataType::Utf8, false),
661        Field::new("metric_value", DataType::Float64, false),
662    ]));
663    let ns_batch = RecordBatch::try_new(
664        ns_schema,
665        vec![
666            Arc::new(StringArray::from(ns_segments)),
667            Arc::new(StringArray::from(ns_variants)),
668            Arc::new(Float64Array::from(ns_values)),
669        ],
670    )?;
671    write_parquet_batch(&output_dir.join("noise_segments_20.parquet"), &ns_batch)?;
672
673    // -------------------------------------------------------------------------
674    // STAT-TRAP-04: spurious_random_walks.parquet
675    // 365 days of two independent positive-drift random walks:
676    //   level Pearson r > 0.85, ADF p > 0.10, first-differenced |r| < 0.15.
677    // -------------------------------------------------------------------------
678    let days: Vec<i64> = (1..=365).collect();
679    let eps_a: Vec<f64> = (0..364).map(|_| rng.next_standard_normal()).collect();
680    let eps_b: Vec<f64> = loop {
681        let candidate: Vec<f64> = (0..364).map(|_| rng.next_standard_normal()).collect();
682        if pearson_r(&eps_a, &candidate).abs() < 0.10 {
683            break candidate;
684        }
685    };
686    let mut series_a = Vec::with_capacity(365);
687    let mut series_b = Vec::with_capacity(365);
688    let mut cur_a = 100.0_f64;
689    let mut cur_b = 50.0_f64;
690    series_a.push(cur_a);
691    series_b.push(cur_b);
692    for i in 0..364 {
693        cur_a += 0.35 + eps_a[i];
694        cur_b += 0.35 + eps_b[i];
695        series_a.push(cur_a);
696        series_b.push(cur_b);
697    }
698    let srw_schema = Arc::new(Schema::new(vec![
699        Field::new("day", DataType::Int64, false),
700        Field::new("series_a", DataType::Float64, false),
701        Field::new("series_b", DataType::Float64, false),
702    ]));
703    let srw_batch = RecordBatch::try_new(
704        srw_schema,
705        vec![
706            Arc::new(Int64Array::from(days)),
707            Arc::new(Float64Array::from(series_a)),
708            Arc::new(Float64Array::from(series_b)),
709        ],
710    )?;
711    write_parquet_batch(
712        &output_dir.join("spurious_random_walks.parquet"),
713        &srw_batch,
714    )?;
715
716    Ok(vec![
717        PlantedTrapSpec {
718            trap_id: "STAT-TRAP-01".to_string(),
719            category: "simpsons_paradox".to_string(),
720            left_dataset: "ab_simpsons_paradox".to_string(),
721            left_field: "converted".to_string(),
722            right_dataset: "ab_simpsons_paradox".to_string(),
723            right_field: "segment".to_string(),
724            expected_action: "stratify_by_segment_and_flag_confounder".to_string(),
725            description: "Aggregate treatment conversion (70%) exceeds control (48%), but treatment loses inside both Enterprise (78% vs 80%) and SMB (38% vs 40%) due to segment mix confounding.".to_string(),
726        },
727        PlantedTrapSpec {
728            trap_id: "STAT-TRAP-02".to_string(),
729            category: "heavy_tailed_outlier_t_test".to_string(),
730            left_dataset: "orders_heavy_tail".to_string(),
731            left_field: "order_total_usd".to_string(),
732            right_dataset: "orders_heavy_tail".to_string(),
733            right_field: "variant".to_string(),
734            expected_action: "detect_normality_failure_and_switch_to_mann_whitney_u".to_string(),
735            description: "Two $50,000 whale orders inflate control mean (~599 vs ~118), reversing naive parametric direction while Mann-Whitney U reveals true treatment median lift (118 vs 100, p < 0.01).".to_string(),
736        },
737        PlantedTrapSpec {
738            trap_id: "STAT-TRAP-03".to_string(),
739            category: "multiple_testing_p_hacking".to_string(),
740            left_dataset: "noise_segments_20".to_string(),
741            left_field: "metric_value".to_string(),
742            right_dataset: "noise_segments_20".to_string(),
743            right_field: "segment_id".to_string(),
744            expected_action: "apply_bh_fdr_and_report_zero_adjusted_discoveries".to_string(),
745            description: "20 independent null segments include >= 1 raw p < 0.05 false positive that disappears to 0 discoveries under Benjamini-Hochberg FDR adjustment.".to_string(),
746        },
747        PlantedTrapSpec {
748            trap_id: "STAT-TRAP-04".to_string(),
749            category: "spurious_nonstationary_correlation".to_string(),
750            left_dataset: "spurious_random_walks".to_string(),
751            left_field: "series_a".to_string(),
752            right_dataset: "spurious_random_walks".to_string(),
753            right_field: "series_b".to_string(),
754            expected_action: "run_adf_kpss_and_first_difference_before_correlation".to_string(),
755            description: "Two independent positive-drift random walks exhibit high spurious level correlation (r > 0.85) that collapses near zero (|r| < 0.15) after first-differencing.".to_string(),
756        },
757    ])
758}

760pub fn seed_validation_suite_with_options(
761    seed: u64,
762    base_row_count: usize,
763    output_dir: &Path,
764    include_stat_traps: bool,
765) -> Result<GroundTruthManifest, SeederError> {
766    let mut manifest = seed_validation_suite(seed, base_row_count, output_dir)?;
767    if include_stat_traps {
768        let stat_traps = seed_statistical_chaos_traps(seed, output_dir)?;
769        manifest.generated_files.extend([
770            "ab_simpsons_paradox.parquet".to_string(),
771            "orders_heavy_tail.parquet".to_string(),
772            "noise_segments_20.parquet".to_string(),
773            "spurious_random_walks.parquet".to_string(),
774        ]);
775        manifest.stat_traps = stat_traps;
776        let manifest_path: PathBuf = output_dir.join("ground_truth_manifest.json");
777        let manifest_json = serde_json::to_string_pretty(&manifest)?;
778        fs::write(manifest_path, manifest_json)?;
779    }
780    Ok(manifest)
781}
