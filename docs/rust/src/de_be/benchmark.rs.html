584
585fn summarize_mode(
586    mode_name: &str,
587    cases: Vec<TrapCaseResult>,
588    provenance_completeness_pct: f64,
589) -> ModeEvaluationSummary {
590    let total_cases = cases.len();
591    let accurate_answers = cases.iter().filter(|c| c.answered_accurately).count();
592    let escalated_to_steward = cases.iter().filter(|c| c.escalated_to_steward).count();
593    let silent_wrong_answers = cases.iter().filter(|c| c.silent_wrong_answer).count();
594
595    let execution_accuracy_pct = if total_cases == 0 {
596        0.0
597    } else {
598        ((accurate_answers as f64) / (total_cases as f64) * 1000.0).round() / 10.0
599    };
600    let silent_error_prevention_pct = if total_cases == 0 {
601        100.0
602    } else {
603        (((total_cases - silent_wrong_answers) as f64) / (total_cases as f64) * 1000.0).round()
604            / 10.0
605    };
606
607    ModeEvaluationSummary {
608        mode_name: mode_name.to_string(),
609        total_cases,
610        accurate_answers,
611        escalated_to_steward,
612        silent_wrong_answers,
613        execution_accuracy_pct,
614        silent_error_prevention_pct,
615        provenance_completeness_pct,
616        cases,
617    }
618}
