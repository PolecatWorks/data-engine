139        &self,
140        id: &str,
141        req: JoinStewardUpdateRequest,
142    ) -> Result<JoinHypothesisRecord, VisualsError> {
143        let mut guard = self.inner.write().map_err(|_| VisualsError::LockPoisoned)?;
144        let existing = guard
145            .joins
146            .get_mut(id)
147            .ok_or_else(|| VisualsError::TopologyFocusNotFound(id.to_string()))?;
148        let status = req.epistemic_status.trim().to_ascii_lowercase();
149        existing.epistemic_status = status.clone();
150        if status == "rejected" {
151            existing.predicate = "FORBIDDEN_JOIN".to_string();
152        } else if status == "verified" && existing.predicate == "FORBIDDEN_JOIN" {
153            existing.predicate = "JOINABLE_WITH".to_string();
154        }
155        if let Some(v) = req.verified_by {
156            existing.verified_by = Some(v);
157        }
158        if let Some(n) = req.steward_notes {
159            existing.steward_notes = Some(n);
160        }
161        if let Some(f) = req.filter_invariant {
162            existing.filter_invariant = Some(f);
163        }
164        Ok(existing.clone())
165    }

415fn normalize_id(raw: &str) -> String {
416    let mut slug = String::with_capacity(raw.len());
417    let mut prev_us = false;
418    for ch in raw.trim().chars() {
419        if ch.is_ascii_alphanumeric() {
420            slug.push(ch.to_ascii_lowercase());
421            prev_us = false;
422        } else if !prev_us && !slug.is_empty() {
423            slug.push('_');
424            prev_us = true;
425        }
426    }
427    while slug.ends_with('_') {
428        slug.pop();
429    }
430    slug
431}

433fn decode_chunked_body(raw: &[u8]) -> Option<Vec<u8>> {
434    let mut out = Vec::new();
435    let mut idx = 0_usize;
436    while idx < raw.len() {
437        let line_end = raw[idx..].windows(2).position(|w| w == b"\r\n")?;
438        let size_str = std::str::from_utf8(&raw[idx..idx + line_end]).ok()?.trim();
439        let hex_part = size_str.split(';').next().unwrap_or(size_str).trim();
440        let chunk_len = usize::from_str_radix(hex_part, 16).ok()?;
441        idx += line_end + 2;
442        if chunk_len == 0 {
443            break;
444        }
445        if idx + chunk_len > raw.len() {
446            return None;
447        }
448        out.extend_from_slice(&raw[idx..idx + chunk_len]);
449        idx += chunk_len;
450        if idx + 2 <= raw.len() && &raw[idx..idx + 2] == b"\r\n" {
451            idx += 2;
452        }
453    }
454    Some(out)
455}

457async fn fetch_graphrag_subgraph(
458    base_url: &str,
459    dataset_ids: &[String],
460    hops: u8,
461    include_forbidden: bool,
462) -> Result<GraphragSubgraphResponse, String> {
463    let stripped = base_url
464        .trim()
465        .strip_prefix("http://")
466        .ok_or_else(|| "only http:// GraphRAG URLs are supported by internal overlay client".to_string())?;
467    let (host_port, base_path) = match stripped.find('/') {
468        Some(pos) => (&stripped[..pos], stripped[pos..].trim_end_matches('/')),
469        None => (stripped, ""),
470    };
471    let addr = if host_port.contains(':') {
472        host_port.to_string()
473    } else {
474        format!("{}:80", host_port)
475    };
476
477    let mut qs = format!("hops={}&include_forbidden={}", hops, include_forbidden);
478    if !dataset_ids.is_empty() {
479        qs.push_str("&dataset_ids=");
480        qs.push_str(&dataset_ids.join(","));
481    }
482    let path_and_query = format!("{}/ontology/subgraph?{}", base_path, qs);
483
484    let fut = async {
485        let mut stream = TcpStream::connect(&addr)
486            .await
487            .map_err(|e| format!("connect failed: {}", e))?;
488        let req = format!(
489            "GET {} HTTP/1.1\r\nHost: {}\r\nAccept: application/json\r\nConnection: close\r\n\r\n",
490            path_and_query, host_port
491        );
492        stream
493            .write_all(req.as_bytes())
494            .await
495            .map_err(|e| format!("write failed: {}", e))?;
496        let mut buf = Vec::new();
497        stream
498            .read_to_end(&mut buf)
499            .await
500            .map_err(|e| format!("read failed: {}", e))?;
501
502        let header_end = buf
503            .windows(4)
504            .position(|w| w == b"\r\n\r\n")
505            .ok_or_else(|| "malformed HTTP response".to_string())?;
506        let header_str = std::str::from_utf8(&buf[..header_end])
507            .map_err(|e| format!("invalid utf8 headers: {}", e))?;
508        let status_line = header_str.lines().next().unwrap_or("");
509        if !status_line.contains(" 200 ") {
510            return Err(format!("non-200 status from Graphrag: {}", status_line));
511        }
512        let is_chunked = header_str
513            .to_ascii_lowercase()
514            .contains("transfer-encoding: chunked");
515        let body_slice = &buf[header_end + 4..];
516        let body_bytes = if is_chunked {
517            decode_chunked_body(body_slice)
518                .ok_or_else(|| "failed to decode chunked HTTP body".to_string())?
519        } else {
520            body_slice.to_vec()
521        };
522        serde_json::from_slice::<GraphragSubgraphResponse>(&body_bytes)
523            .map_err(|e| format!("invalid Graphrag JSON: {}", e))
524    };
525
526    tokio::time::timeout(Duration::from_secs(2), fut)
527        .await
528        .map_err(|_| "Graphrag subgraph request timed out".to_string())?
529}

544/// Builds a fused, bounded multi-layer `SemanticTopologyGraph` combining `data-engine`'s catalog,
545/// statistical field traits/confounders, provenance lineage, and `Graphrag`'s Layer 2 ontology.
546pub async fn build_topology_graph(
547    catalog: &TopologyCatalog,
548    field_catalog: &FieldStatisticalCatalog,
549    provenance_store: &ProvenanceStore,
550    config: &VisualsConfig,
551    query: &TopologyGraphQuery,
552) -> Result<SemanticTopologyGraph, VisualsError> {
553    let lens = query
554        .lens
555        .clone()
556        .unwrap_or_else(|| "semantic_3layer".to_string());
557    let hops = query.hops.unwrap_or(config.default_ego_hops);
558    let include_forbidden = query.include_forbidden.unwrap_or(true);
559    let include_semantics = query.include_semantics.unwrap_or(true);
560
561    let all_datasets = catalog.list_datasets()?;
562    let all_joins = catalog.list_joins()?;
563    let all_traits = field_catalog.list().unwrap_or_default();
564    let trait_map: HashMap<String, LearnedFieldTrait> = all_traits
565        .into_iter()
566        .map(|t| (t.field_canonical_id.clone(), t))
567        .collect();
568
569    let dataset_filter: Option<HashSet<String>> = query
570        .dataset_ids
571        .as_ref()
572        .map(|v| v.iter().map(|s| normalize_id(s)).collect());
573
574    let scoped_datasets: Vec<CatalogDatasetRecord> = all_datasets
575        .into_iter()
576        .filter(|ds| {
577            if let Some(ref src) = query.source_id
578                && ds.source_name != *src
579            {
580                return false;
581            }
582            if let Some(ref set) = dataset_filter
583                && !set.contains(&ds.canonical_id)
584            {
585                return false;
586            }
587            true
588        })
589        .collect();
590
591    let include_fields = query
592        .include_fields
593        .unwrap_or(scoped_datasets.len() <= 15);
594
595    let mut nodes_map: BTreeMap<String, TopologyNode> = BTreeMap::new();
596    let mut edges_vec: Vec<TopologyEdge> = Vec::new();
597    let mut communities: Vec<TopologyCommunity> = Vec::new();
598
599    // 1. Add Layer 1 DataSource & Dataset nodes + Layer 0 DatasetField nodes
600    let mut source_ids_seen: HashSet<String> = HashSet::new();
601    for ds in &scoped_datasets {
602        let src_node_id = format!("src_{}", normalize_id(&ds.source_name));
603        if source_ids_seen.insert(src_node_id.clone()) {
604            nodes_map.insert(
605                src_node_id.clone(),
606                TopologyNode {
607                    id: src_node_id.clone(),
608                    kind: "source".to_string(),
609                    layer: 1,
610                    parent_id: None,
611                    label: ds.source_name.clone(),
612                    subtitle: Some("Registered Data Source".to_string()),
613                    epistemic_status: "verified".to_string(),
614                    confidence: 1.0,
615                    badges: vec!["source".to_string()],
616                    field_count: None,
617                    metadata: json!({ "source_name": ds.source_name }),
618                },
619            );
620        }
621
622        let mut ds_badges = vec![ds.format.clone()];
623        if ds.profile.epistemic_status == "drifted" || !ds.profile.drift_findings.is_empty() {
624            ds_badges.push("drifted".to_string());
625        }
626        let ds_status = if !ds.profile.drift_findings.is_empty() {
627            "drifted".to_string()
628        } else {
629            ds.profile.epistemic_status.clone()
630        };
631
632        nodes_map.insert(
633            ds.canonical_id.clone(),
634            TopologyNode {
635                id: ds.canonical_id.clone(),
636                kind: "dataset".to_string(),
637                layer: 1,
638                parent_id: Some(src_node_id.clone()),
639                label: ds.profile.dataset_name.clone(),
640                subtitle: Some(format!(
641                    "{} • {} rows",
642                    ds.grain_description, ds.profile.row_count
643                )),
644                epistemic_status: ds_status,
645                confidence: 1.0,
646                badges: ds_badges,
647                field_count: Some(ds.profile.fields.len()),
648                metadata: json!({
649                    "format": ds.format,
650                    "row_count_estimate": ds.profile.row_count,
651                    "grain_description": ds.grain_description,
652                    "schema_fingerprint_sha256": ds.profile.schema_fingerprint_sha256,
653                    "drift_findings": ds.profile.drift_findings,
654                }),
655            },
656        );
657        edges_vec.push(TopologyEdge {
658            id: format!("provides_{}_{}", src_node_id, ds.canonical_id),
659            source_id: src_node_id,
660            target_id: ds.canonical_id.clone(),
661            predicate: "PROVIDES".to_string(),
662            cardinality: None,
663            epistemic_status: "verified".to_string(),
664            confidence: 1.0,
665            inclusion_ratio: None,
666            fan_out_warning: None,
667            filter_invariant: None,
668            steward_notes: None,
669        });
670
671        let drifted_fields: HashSet<&str> = ds
672            .profile
673            .drift_findings
674            .iter()
675            .map(|d| d.field_name.as_str())
676            .collect();
677
678        if include_fields {
679            for field in &ds.profile.fields {
680                let canonical_field = format!("{}.{}", ds.canonical_id, field.field_name);
681                let stat_trait = trait_map.get(&canonical_field);
682                let mut f_badges = vec![field.arrow_data_type.clone()];
683                if ds
684                    .profile
685                    .candidate_primary_keys
686                    .contains(&field.field_name)
687                {
688                    f_badges.push("pk".to_string());
689                }
690                let f_status = if drifted_fields.contains(field.field_name.as_str()) {
691                    f_badges.push("drifted".to_string());
692                    "drifted".to_string()
693                } else {
694                    field.epistemic_status.clone()
695                };
696                if let Some(tr) = stat_trait {
697                    if tr.is_heavy_tailed {
698                        f_badges.push("heavy_tail".to_string());
699                    }
700                    if tr.seasonality_period.is_some() {
701                        f_badges.push("seasonal".to_string());
702                    }
703                }
704                nodes_map.insert(
705                    canonical_field.clone(),
706                    TopologyNode {
707                        id: canonical_field.clone(),
708                        kind: "field".to_string(),
709                        layer: 0,
710                        parent_id: Some(ds.canonical_id.clone()),
711                        label: field.field_name.clone(),
712                        subtitle: Some(format!(
713                            "{} • {:.0}% null",
714                            field.arrow_data_type,
715                            field.null_ratio * 100.0
716                        )),
717                        epistemic_status: f_status,
718                        confidence: stat_trait.map(|t| t.confidence).unwrap_or(1.0),
719                        badges: f_badges,
720                        field_count: None,
721                        metadata: json!({
722                            "arrow_data_type": field.arrow_data_type,
723                            "null_ratio": field.null_ratio,
724                            "distinct_count": field.distinct_count,
725                            "inferred_semantic_domain": field.inferred_semantic_domain,
726                            "statistical_profile": stat_trait,
727                        }),
728                    },
729                );
730                edges_vec.push(TopologyEdge {
731                    id: format!("has_field_{}", canonical_field),
732                    source_id: ds.canonical_id.clone(),
733                    target_id: canonical_field.clone(),
734                    predicate: "HAS_FIELD".to_string(),
735                    cardinality: None,
736                    epistemic_status: "verified".to_string(),
737                    confidence: 1.0,
738                    inclusion_ratio: None,
739                    fan_out_warning: None,
740                    filter_invariant: None,
741                    steward_notes: None,
742                });
743
744                // Emit local CONFOUNDED_BY edges from FieldStatisticalCatalog
745                if let Some(tr) = stat_trait {
746                    for conf in &tr.known_confounders {
747                        edges_vec.push(TopologyEdge {
748                            id: format!("confounded_{}_{}", canonical_field, conf),
749                            source_id: canonical_field.clone(),
750                            target_id: conf.clone(),
751                            predicate: "CONFOUNDED_BY".to_string(),
752                            cardinality: None,
753                            epistemic_status: tr.epistemic_status.clone(),
754                            confidence: tr.confidence,
755                            inclusion_ratio: None,
756                            fan_out_warning: None,
757                            filter_invariant: None,
758                            steward_notes: Some("Statistical confounder (Simpson's paradox guard)".to_string()),
759                        });
760                    }
761                }
762            }
763        }
764    }
765
766    // 2. Count outgoing 1:N / N:M joins per dataset to detect TRAP-03 multi-child chasm fan-outs
767    let mut multi_cardinality_counts: HashMap<String, usize> = HashMap::new();
768    for j in &all_joins {
769        if j.predicate == "JOINABLE_WITH"
770            && (j.probe.observed_cardinality == "1:N"
771                || j.probe.observed_cardinality == "N:M"
772                || j.probe.fan_out_risk)
773        {
774            *multi_cardinality_counts
775                .entry(normalize_id(&j.probe.left_dataset))
776                .or_insert(0) += 1;
777        }
778    }
779
780    for j in &all_joins {
781        if !include_forbidden && j.predicate == "FORBIDDEN_JOIN" {
782            continue;
783        }
784        let left_ds = normalize_id(&j.probe.left_dataset);
785        let right_ds = normalize_id(&j.probe.right_dataset);
786        let left_endpoint = if include_fields {
787            format!("{}.{}", left_ds, j.probe.left_field)
788        } else {
789            left_ds.clone()
790        };
791        let right_endpoint = if include_fields {
792            format!("{}.{}", right_ds, j.probe.right_field)
793        } else {
794            right_ds.clone()
795        };
796        if !nodes_map.contains_key(&left_endpoint) || !nodes_map.contains_key(&right_endpoint) {
797            continue;
798        }
799
800        let is_multi_chasm =
801            multi_cardinality_counts.get(&left_ds).copied().unwrap_or(0) >= 2;
802        let fan_out_warning = if j.predicate == "JOINABLE_WITH"
803            && (j.probe.fan_out_risk
804                || j.probe.observed_cardinality == "N:M"
805                || is_multi_chasm)
806        {
807            Some(FanOutWarning {
808                multiplier_estimate: j.probe.row_multiplication_factor.max(1.5),
809                reason: if is_multi_chasm {
810                    format!(
811                        "Chasm fan-out risk: '{}' joins to multiple 1:N child tables; pre-aggregate before joining.",
812                        left_ds
813                    )
814                } else {
815                    format!(
816                        "Fan-out risk ({} cardinality, {:.2}x row multiplication).",
817                        j.probe.observed_cardinality, j.probe.row_multiplication_factor
818                    )
819                },
820                suggested_cte_pattern: format!(
821                    "WITH agg_{r} AS (SELECT {rf}, SUM(metric) AS total FROM {r} GROUP BY {rf}) SELECT ... FROM {l} JOIN agg_{r} ON {l}.{lf} = agg_{r}.{rf}",
822                    l = left_ds,
823                    lf = j.probe.left_field,
824                    r = right_ds,
825                    rf = j.probe.right_field
826                ),
827            })
828        } else {
829            None
830        };
831
832        edges_vec.push(TopologyEdge {
833            id: j.id.clone(),
834            source_id: left_endpoint,
835            target_id: right_endpoint,
836            predicate: j.predicate.clone(),
837            cardinality: Some(j.probe.observed_cardinality.clone()),
838            epistemic_status: j.epistemic_status.clone(),
839            confidence: j.probe.confidence,
840            inclusion_ratio: Some(j.probe.inclusion_ratio_left_in_right),
841            fan_out_warning,
842            filter_invariant: j.filter_invariant.clone(),
843            steward_notes: j.steward_notes.clone(),
844        });
845    }
846
847    // 3. Fuse GraphRAG Layer 2 semantic ontology overlay (with graceful degradation)
848    let mut graphrag_overlay_degraded = false;
849    if include_semantics && config.enable_graphrag_overlay {
850        if let Some(ref base_url) = config.graphrag_base_url {
851            let ds_names: Vec<String> = scoped_datasets
852                .iter()
853                .map(|d| d.canonical_id.clone())
854                .collect();
855            match fetch_graphrag_subgraph(base_url, &ds_names, hops, include_forbidden).await {
856                Ok(gr) => {
857                    // Build lookup from GraphRAG field node id -> canonical "ds.field" id
858                    let mut gr_id_remap: HashMap<String, String> = HashMap::new();
859                    for gn in &gr.nodes {
860                        if let Some(canon) = gn
861                            .properties
862                            .get("canonical_id")
863                            .and_then(|v| v.as_str())
864                            && nodes_map.contains_key(canon)
865                        {
866                            gr_id_remap.insert(gn.id.clone(), canon.to_string());
867                        }
868                    }
869
870                    for gn in gr.nodes {
871                        let mapped_id = gr_id_remap
872                            .get(&gn.id)
873                            .cloned()
874                            .unwrap_or_else(|| gn.id.clone());
875                        if nodes_map.contains_key(&mapped_id) {
876                            continue;
877                        }
878                        let kind = match gn.label.as_str() {
879                            "BusinessConcept" => "business_concept",
880                            "BusinessMetric" => "business_metric",
881                            "CommunitySummary" => "community_summary",
882                            "Dataset" => "dataset",
883                            "Field" => "field",
884                            _ => "business_concept",
885                        };
886                        nodes_map.insert(
887                            mapped_id.clone(),
888                            TopologyNode {
889                                id: mapped_id,
890                                kind: kind.to_string(),
891                                layer: gn.layer,
892                                parent_id: None,
893                                label: gn.id,
894                                subtitle: Some(gn.description),
895                                epistemic_status: gn.epistemic_status,
896                                confidence: gn.confidence,
897                                badges: vec![gn.label],
898                                field_count: None,
899                                metadata: json!({
900                                    "verified_by": gn.verified_by,
901                                    "properties": gn.properties,
902                                }),
903                            },
904                        );
905                    }
906
907                    for ge in gr.edges {
908                        let src = gr_id_remap
909                            .get(&ge.source_id)
910                            .cloned()
911                            .unwrap_or(ge.source_id);
912                        let tgt = gr_id_remap
913                            .get(&ge.target_id)
914                            .cloned()
915                            .unwrap_or(ge.target_id);
916                        if !nodes_map.contains_key(&src) || !nodes_map.contains_key(&tgt) {
917                            continue;
918                        }
919                        let edge_id = format!("gr_{}_{}_{}", src, ge.predicate, tgt);
920                        if edges_vec
921                            .iter()
922                            .any(|e| e.source_id == src && e.predicate == ge.predicate && e.target_id == tgt)
923                        {
924                            continue;
925                        }
926                        edges_vec.push(TopologyEdge {
927                            id: edge_id,
928                            source_id: src,
929                            target_id: tgt,
930                            predicate: ge.predicate,
931                            cardinality: ge
932                                .properties
933                                .get("cardinality")
934                                .and_then(|v| v.as_str())
935                                .map(str::to_string),
936                            epistemic_status: ge.epistemic_status,
937                            confidence: ge.confidence,
938                            inclusion_ratio: ge
939                                .properties
940                                .get("inclusion_ratio")
941                                .and_then(|v| v.as_f64()),
942                            fan_out_warning: None,
943                            filter_invariant: None,
944                            steward_notes: ge
945                                .properties
946                                .get("rejection_reason")
947                                .and_then(|v| v.as_str())
948                                .map(str::to_string),
949                        });
950                    }
951
952                    for gc in gr.communities {
953                        communities.push(TopologyCommunity {
954                            id: gc.id,
955                            label: gc.label,
956                            member_node_ids: gc
957                                .member_node_ids
958                                .into_iter()
959                                .map(|m| gr_id_remap.get(&m).cloned().unwrap_or(m))
960                                .collect(),
961                        });
962                    }
963                }
964                Err(_) => {
965                    graphrag_overlay_degraded = true;
966                }
967            }
968        }
969    }
970
971    // 4. Optional focus ego-network BFS
972    if let Some(ref focus) = query.focus_node_id {
973        let focus_key = if nodes_map.contains_key(focus) {
974            focus.clone()
975        } else {
976            let norm = normalize_id(focus);
977            if nodes_map.contains_key(&norm) {
978                norm
979            } else {
980                return Err(VisualsError::TopologyFocusNotFound(focus.clone()));
981            }
982        };
983        let mut adj: HashMap<&str, Vec<&str>> = HashMap::new();
984        for e in &edges_vec {
985            adj.entry(e.source_id.as_str()).or_default().push(e.target_id.as_str());
986            adj.entry(e.target_id.as_str()).or_default().push(e.source_id.as_str());
987        }
988        let mut visited: HashSet<String> = HashSet::new();
989        let mut q: VecDeque<(String, u8)> = VecDeque::new();
990        visited.insert(focus_key.clone());
991        q.push_back((focus_key, 0));
992        while let Some((curr, dist)) = q.pop_front() {
993            if dist >= hops {
994                continue;
995            }
996            if let Some(nbrs) = adj.get(curr.as_str()) {
997                for &n in nbrs {
998                    if visited.insert(n.to_string()) {
999                        q.push_back((n.to_string(), dist + 1));
1000                    }
1001                }
1002            }
1003        }
1004        nodes_map.retain(|k, _| visited.contains(k));
1005    }
1006
1007    // 5. Optional epistemic_status filter
1008    if let Some(ref statuses) = query.epistemic_status
1009        && !statuses.is_empty()
1010    {
1011        let allowed: HashSet<String> = statuses
1012            .iter()
1013            .map(|s| s.trim().to_ascii_lowercase())
1014            .collect();
1015        nodes_map.retain(|_, n| {
1016            n.kind == "source" || allowed.contains(&n.epistemic_status.to_ascii_lowercase())
1017        });
1018    }
1019
1020    // 6. Deterministic sort + bounds enforcement (zero dangling edges)
1021    let mut nodes: Vec<TopologyNode> = nodes_map.into_values().collect();
1022    nodes.sort_by(|a, b| {
1023        (Reverse(a.layer), epistemic_priority(&a.epistemic_status), &a.id).cmp(&(
1024            Reverse(b.layer),
1025            epistemic_priority(&b.epistemic_status),
1026            &b.id,
1027        ))
1028    });
1029    let total_nodes = nodes.len();
1030    let nodes_truncated = total_nodes > config.max_topology_nodes;
1031    nodes.truncate(config.max_topology_nodes);
1032
1033    let kept_ids: HashSet<&str> = nodes.iter().map(|n| n.id.as_str()).collect();
1034    let mut filtered_edges: Vec<TopologyEdge> = edges_vec
1035        .into_iter()
1036        .filter(|e| {
1037            kept_ids.contains(e.source_id.as_str()) && kept_ids.contains(e.target_id.as_str())
1038        })
1039        .collect();
1040    filtered_edges.sort_by(|a, b| {
1041        (&a.source_id, &a.predicate, &a.target_id).cmp(&(&b.source_id, &b.predicate, &b.target_id))
1042    });
1043    let total_edges = filtered_edges.len();
1044    let edges_truncated = total_edges > config.max_topology_edges;
1045    filtered_edges.truncate(config.max_topology_edges);
1046
1047    let truncated = nodes_truncated || edges_truncated;
1048    let applied_filter_hint = if truncated {
1049        Some("Graph exceeded configured bounds; pass focus_node_id, dataset_ids, or include_fields=false to narrow scope.".to_string())
1050    } else {
1051        None
1052    };
1053
1054    // 7. Optional ProvenanceRun lineage overlay highlight
1055    let highlighted_path = if let Some(ref run_id) = query.provenance_run_id {
1056        if let Ok(run) = provenance_store.get_run(run_id) {
1057            let mut h_nodes: Vec<String> = Vec::new();
1058            for ds in &run.datasets_consulted {
1059                let norm = normalize_id(ds);
1060                if kept_ids.contains(norm.as_str()) {
1061                    h_nodes.push(norm);
1062                }
1063            }
1064            let h_set: HashSet<&str> = h_nodes.iter().map(String::as_str).collect();
1065            let h_edges: Vec<String> = filtered_edges
1066                .iter()
1067                .filter(|e| h_set.contains(e.source_id.as_str()) || h_set.contains(e.target_id.as_str()))
1068                .filter(|e| e.predicate == "JOINABLE_WITH" || e.predicate == "COMPUTED_FROM")
1069                .map(|e| e.id.clone())
1070                .collect();
1071            let warnings: Vec<String> = filtered_edges
1072                .iter()
1073                .filter(|e| h_edges.contains(&e.id))
1074                .filter_map(|e| e.fan_out_warning.as_ref().map(|w| w.reason.clone()))
1075                .collect();
1076            Some(HighlightedPath {
1077                node_ids: h_nodes,
1078                edge_ids: h_edges,
1079                all_rules_human_verified: run.trust_summary.all_rules_human_verified,
1080                fan_out_warnings: warnings,
1081            })
1082        } else {
1083            None
1084        }
1085    } else {
1086        None
1087    };
1088
1089    Ok(SemanticTopologyGraph {
1090        lens,
1091        graphrag_overlay_degraded,
1092        truncated,
1093        total_nodes,
1094        total_edges,
1095        applied_filter_hint,
1096        communities,
1097        nodes,
1098        edges: filtered_edges,
1099        highlighted_path,
1100    })
1101}

1103fn insert_schema_path(
1104    roots: &mut Vec<SchemaTreeNode>,
1105    segments: &[&str],
1106    prefix: &str,
1107    field: &FieldProfile,
1108    is_pk: bool,
1109    stat_trait: Option<LearnedFieldTrait>,
1110) {
1111    if segments.is_empty() {
1112        return;
1113    }
1114    let seg = segments[0];
1115    let is_list = seg.ends_with("[]");
1116    let full_path = if prefix.is_empty() {
1117        seg.to_string()
1118    } else {
1119        format!("{}.{}", prefix, seg)
1120    };
1121
1122    if segments.len() == 1 {
1123        let bins: Vec<MiniHistogramBin> = field
1124            .top_k_values
1125            .iter()
1126            .take(5)
1127            .enumerate()
1128            .map(|(i, v)| MiniHistogramBin {
1129                label: v.clone(),
1130                count: (5 - i) * 10,
1131            })
1132            .collect();
1133        roots.push(SchemaTreeNode {
1134            name: seg.to_string(),
1135            full_path,
1136            node_kind: "field".to_string(),
1137            arrow_data_type: Some(field.arrow_data_type.clone()),
1138            is_primary_key_candidate: is_pk,
1139            epistemic_status: Some(field.epistemic_status.clone()),
1140            null_ratio: Some(field.null_ratio),
1141            distinct_count: Some(field.distinct_count),
1142            top_values: field.top_k_values.clone(),
1143            mini_histogram_bins: bins,
1144            statistical_profile: stat_trait,
1145            children: vec![],
1146        });
1147        return;
1148    }
1149
1150    let existing_pos = roots.iter().position(|n| n.name == seg && n.node_kind != "field");
1151    let container_idx = if let Some(pos) = existing_pos {
1152        pos
1153    } else {
1154        roots.push(SchemaTreeNode {
1155            name: seg.to_string(),
1156            full_path: full_path.clone(),
1157            node_kind: if is_list { "list" } else { "struct" }.to_string(),
1158            arrow_data_type: Some(if is_list { "List<Struct>" } else { "Struct" }.to_string()),
1159            is_primary_key_candidate: false,
1160            epistemic_status: None,
1161            null_ratio: None,
1162            distinct_count: None,
1163            top_values: vec![],
1164            mini_histogram_bins: vec![],
1165            statistical_profile: None,
1166            children: vec![],
1167        });
1168        roots.len() - 1
1169    };
1170
1171    insert_schema_path(
1172        &mut roots[container_idx].children,
1173        &segments[1..],
1174        &full_path,
1175        field,
1176        is_pk,
1177        stat_trait,
1178    );
1179}

1181/// Builds zoomed-in structural anatomy for one dataset (nested Arrow schema tree + profile bins + incident joins).
1182pub fn build_dataset_anatomy(
1183    dataset_id_or_name: &str,
1184    catalog: &TopologyCatalog,
1185    field_catalog: &FieldStatisticalCatalog,
1186) -> Result<DatasetAnatomyResponse, VisualsError> {
1187    let ds = catalog
1188        .get_dataset(dataset_id_or_name)?
1189        .ok_or_else(|| VisualsError::TopologyFocusNotFound(dataset_id_or_name.to_string()))?;
1190
1191    let mut schema_tree: Vec<SchemaTreeNode> = Vec::new();
1192    for field in &ds.profile.fields {
1193        let canonical = format!("{}.{}", ds.canonical_id, field.field_name);
1194        let stat_trait = field_catalog.get(&canonical).ok().flatten();
1195        let is_pk = ds.profile.candidate_primary_keys.contains(&field.field_name);
1196        let segments: Vec<&str> = field.field_name.split('.').collect();
1197        insert_schema_path(&mut schema_tree, &segments, "", field, is_pk, stat_trait);
1198    }
1199
1200    let incident_joins: Vec<JoinHypothesisRecord> = catalog
1201        .list_joins()?
1202        .into_iter()
1203        .filter(|j| {
1204            normalize_id(&j.probe.left_dataset) == ds.canonical_id
1205                || normalize_id(&j.probe.right_dataset) == ds.canonical_id
1206        })
1207        .collect();
1208
1209    Ok(DatasetAnatomyResponse {
1210        dataset_id: ds.id,
1211        canonical_id: ds.canonical_id,
1212        format: ds.format,
1213        grain_description: ds.grain_description,
1214        row_count_estimate: ds.profile.row_count,
1215        schema_fingerprint_sha256: ds.profile.schema_fingerprint_sha256,
1216        epistemic_status: ds.profile.epistemic_status,
1217        schema_tree,
1218        incident_joins,
1219        semantic_links: vec![],
1220    })
1221}

1226    fallback_row_count: usize,
1227    catalog: &TopologyCatalog,
1228) -> (SchemaCard, String) {
1229    if let Ok(Some(ds)) = catalog.get_dataset(dataset_name) {
1230        let mut join_type = "Unknown".to_string();
1231        let fields = ds
1232            .profile
1233            .fields
1234            .iter()
1235            .map(|f| {
1236                let is_key = f.field_name == join_field;
1237                if is_key {
1238                    join_type = f.arrow_data_type.clone();
1239                }
1240                SchemaCardField {
1241                    field_name: f.field_name.clone(),
1242                    arrow_data_type: f.arrow_data_type.clone(),
1243                    is_join_key: is_key,
1244                    is_primary_key_candidate: ds
1245                        .profile
1246                        .candidate_primary_keys
1247                        .contains(&f.field_name),
1248                    null_ratio: f.null_ratio,
1249                    distinct_count: f.distinct_count,
1250                    epistemic_status: f.epistemic_status.clone(),
1251                }
1252            })
1253            .collect();
1254        (
1255            SchemaCard {
1256                dataset_id: ds.canonical_id,
1257                row_count: ds.profile.row_count,
1258                join_field: join_field.to_string(),
1259                fields,
1260            },
1261            join_type,
1262        )

1284/// Builds zoomed-in Field-to-Field Join Anatomy with schema cards, Arrow type compatibility,
1285/// Venn/Sankey overlap strip, fan-out diagnostics, and ready-to-run `explore_sql_template`.
1286pub fn build_join_anatomy(
1287    join_id: &str,
1288    catalog: &TopologyCatalog,
1289) -> Result<JoinAnatomyResponse, VisualsError> {
1290    let join = catalog
1291        .get_join(join_id)?
1292        .ok_or_else(|| VisualsError::TopologyFocusNotFound(join_id.to_string()))?;
1293
1294    let (left_card, left_type) = build_schema_card(
1295        &join.probe.left_dataset,
1296        &join.probe.left_field,
1297        join.probe.left_row_count,
1298        catalog,
1299    );
1300    let (right_card, right_type) = build_schema_card(
1301        &join.probe.right_dataset,
1302        &join.probe.right_field,
1303        join.probe.right_row_count,
1304        catalog,
1305    );
1306
1307    let compatible = left_type == right_type || left_type == "Unknown" || right_type == "Unknown";
1308    let (cast_required, type_warning) = if compatible {
1309        (None, None)
1310    } else {
1311        (
1312            Some(format!("CAST(l.{} AS {})", join.probe.left_field, right_type)),
1313            Some(format!(
1314                "Arrow type mismatch: {}.{} is '{}' while {}.{} is '{}'; explicit cast required.",
1315                join.probe.left_dataset,
1316                join.probe.left_field,
1317                left_type,
1318                join.probe.right_dataset,
1319                join.probe.right_field,
1320                right_type
1321            )),
1322        )
1323    };
1324
1325    let matched_distinct = ((join.probe.left_distinct_count as f64)
1326        * join.probe.inclusion_ratio_left_in_right)
1327        .round() as usize;
1328    let matched_distinct = matched_distinct
1329        .min(join.probe.left_distinct_count)
1330        .min(join.probe.right_distinct_count);
1331    let left_orphans = join
1332        .probe
1333        .left_distinct_count
1334        .saturating_sub(matched_distinct);
1335    let right_orphans = join
1336        .probe
1337        .right_distinct_count
1338        .saturating_sub(matched_distinct);
1339
1340    let is_fan_out = join.probe.fan_out_risk || join.probe.observed_cardinality == "N:M";
1341    let safe_cte_sql_hint = if is_fan_out {
1342        Some(format!(
1343            "WITH right_agg AS (SELECT {rf}, COUNT(*) AS right_rows FROM {r} GROUP BY {rf}) SELECT l.*, right_agg.right_rows FROM {l} l LEFT JOIN right_agg ON l.{lf} = right_agg.{rf}",
1344            l = left_card.dataset_id,
1345            lf = join.probe.left_field,
1346            r = right_card.dataset_id,
1347            rf = join.probe.right_field
1348        ))
1349    } else {
1350        None
1351    };
1352
1353    let explore_sql_template = format!(
1354        "SELECT l.*, r.* FROM {} l JOIN {} r ON l.{} = r.{} LIMIT 1000",
1355        left_card.dataset_id,
1356        right_card.dataset_id,
1357        join.probe.left_field,
1358        join.probe.right_field
1359    );
1360
1361    Ok(JoinAnatomyResponse {
1362        left_dataset: left_card,
1363        right_dataset: right_card,
1364        type_compatibility: TypeCompatibility {
1365            compatible,
1366            left_arrow_type: left_type,
1367            right_arrow_type: right_type,
1368            cast_required,
1369            warning: type_warning,
1370        },
1371        overlap_strip: OverlapStrip {
1372            jaccard_overlap: join.probe.jaccard_similarity,
1373            left_inclusion_ratio: join.probe.inclusion_ratio_left_in_right,
1374            right_inclusion_ratio: join.probe.inclusion_ratio_right_in_left,
1375            left_distinct_count: join.probe.left_distinct_count,
1376            right_distinct_count: join.probe.right_distinct_count,
1377            matched_distinct_count: matched_distinct,
1378            left_orphan_count: left_orphans,
1379            right_orphan_count: right_orphans,
1380        },
1381        matched_samples: vec![],
1382        left_orphan_samples: vec![],
1383        right_orphan_samples: vec![],
1384        fan_out_diagnostics: FanOutDiagnostics {
1385            is_fan_out_risk: is_fan_out,
1386            row_multiplier_estimate: join.probe.row_multiplication_factor,
1387            explanation: join.probe.explanation.clone(),
1388            safe_cte_sql_hint,
1389        },
1390        explore_sql_template,
1391        join,
1392    })
1393}

1395/// Deterministically renders a scoped `SemanticTopologyGraph` into Mermaid `erDiagram` or `flowchart TB`.
1396pub fn export_topology_mermaid(graph: SemanticTopologyGraph) -> TopologyMermaidResponse {
1397    if graph.lens == "er_join" {
1398        let mut lines = vec!["erDiagram".to_string()];
1399        let datasets: Vec<&TopologyNode> = graph
1400            .nodes
1401            .iter()
1402            .filter(|n| n.kind == "dataset")
1403            .collect();
1404        for ds in &datasets {
1405            let entity_id = normalize_id(&ds.id).to_ascii_uppercase();
1406            lines.push(format!("    {} {{", entity_id));
1407            for field in graph
1408                .nodes
1409                .iter()
1410                .filter(|f| f.kind == "field" && f.parent_id.as_deref() == Some(ds.id.as_str()))
1411            {
1412                let dt = field
1413                    .metadata
1414                    .get("arrow_data_type")
1415                    .and_then(|v| v.as_str())
1416                    .unwrap_or("string");
1417                let pk = if field.badges.iter().any(|b| b == "pk") {
1418                    " PK"
1419                } else {
1420                    ""
1421                };
1422                lines.push(format!("        {} {}{}", dt, field.label, pk));
1423            }
1424            lines.push("    }".to_string());
1425        }
1426        for e in &graph.edges {
1427            if e.predicate != "JOINABLE_WITH" && e.predicate != "FORBIDDEN_JOIN" {
1428                continue;
1429            }
1430            let left_ds = e
1431                .source_id
1432                .split('.')
1433                .next()
1434                .unwrap_or(&e.source_id)
1435                .to_ascii_uppercase();
1436            let right_ds = e
1437                .target_id
1438                .split('.')
1439                .next()
1440                .unwrap_or(&e.target_id)
1441                .to_ascii_uppercase();
1442            if e.predicate == "FORBIDDEN_JOIN" {
1443                lines.push(format!(
1444                    "    %% FORBIDDEN_JOIN: {} x {} ({})",
1445                    e.source_id,
1446                    e.target_id,
1447                    e.steward_notes.as_deref().unwrap_or("rejected")
1448                ));
1449            } else {
1450                let rel = match e.cardinality.as_deref() {
1451                    Some("1:1") => "||--||",
1452                    Some("1:N") => "||--o{",
1453                    Some("N:1") => "}o--||",
1454                    _ => "}o--o{",
1455                };
1456                lines.push(format!(
1457                    "    {} {} {} : \"{} ({})\"",
1458                    left_ds,
1459                    rel,
1460                    right_ds,
1461                    e.cardinality.as_deref().unwrap_or("join"),
1462                    e.epistemic_status
1463                ));
1464            }
1465        }
1466        TopologyMermaidResponse {
1467            diagram_kind: "erDiagram".to_string(),
1468            mermaid: lines.join("\n"),
1469            graph,
1470        }
1471    } else {
1472        let mut lines = vec!["flowchart TB".to_string()];
1473        let mut l2 = Vec::new();
1474        let mut l1 = Vec::new();
1475        let mut l0 = Vec::new();
1476        for n in &graph.nodes {
1477            let safe_id = format!("n_{}", normalize_id(&n.id));
1478            let label = format!("{} [{}]", n.label, n.epistemic_status);
1479            let line = format!("        {}[\"{}\"]", safe_id, label);
1480            match n.layer {
1481                2 => l2.push(line),
1482                1 => l1.push(line),
1483                _ => l0.push(line),
1484            }
1485        }
1486        if !l2.is_empty() {
1487            lines.push("    subgraph Layer2[\"Layer 2: Semantic Ontology\"]".to_string());
1488            lines.extend(l2);
1489            lines.push("    end".to_string());
1490        }
1491        if !l1.is_empty() {
1492            lines.push("    subgraph Layer1[\"Layer 1: Datasets\"]".to_string());
1493            lines.extend(l1);
1494            lines.push("    end".to_string());
1495        }
1496        if !l0.is_empty() {
1497            lines.push("    subgraph Layer0[\"Layer 0: Physical Fields\"]".to_string());
1498            lines.extend(l0);
1499            lines.push("    end".to_string());
1500        }
1501        for e in &graph.edges {
1502            let s = format!("n_{}", normalize_id(&e.source_id));
1503            let t = format!("n_{}", normalize_id(&e.target_id));
1504            let arrow = if e.predicate == "FORBIDDEN_JOIN" || e.predicate == "CONFOUNDED_BY" {
1505                format!("-. \"{} [{}]\" .->", e.predicate, e.epistemic_status)
1506            } else {
1507                format!("-->|\"{} [{}]\"|", e.predicate, e.epistemic_status)
1508            };
1509            lines.push(format!("    {} {} {}", s, arrow, t));
1510        }
1511        TopologyMermaidResponse {
1512            diagram_kind: "flowchart".to_string(),
1513            mermaid: lines.join("\n"),
1514            graph,
1515        }
1516    }
1517}
