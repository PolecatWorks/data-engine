18pub const TRAP_05_ZSCORE_CHURN_PYTHON_CODE: &str = r#"
19import math
20import statistics
21
22def compute(rows):
23    totals = [float(r["order_total_usd"]) for r in rows]
24    n = len(totals)
25    if n == 0:
26        return {"anomaly_count": 0, "anomalous_order_ids": [], "mean_usd": 0.0, "std_usd": 0.0}
27    mean_val = sum(totals) / float(n)
28    variance = sum((x - mean_val) ** 2 for x in totals) / float(n)
29    std_val = math.sqrt(variance)
30
31    anomalous_orders = []
32    customer_risk = {}
33    for r in rows:
34        amt = float(r["order_total_usd"])
35        z = (amt - mean_val) / std_val if std_val > 0.0 else 0.0
36        if z > 3.0:
37            anomalous_orders.append({
38                "order_id": r["order_id"],
39                "cust_ref_key": r["cust_ref_key"],
40                "order_total_usd": round(amt, 2),
41                "z_score": round(z, 3)
42            })
43        cust = r["cust_ref_key"]
44        status_weight = 1.25 if r.get("status") == "shipped" else 1.0
45        customer_risk[cust] = round(customer_risk.get(cust, 0.0) + max(z, 0.0) * status_weight, 4)
46
47    return {
48        "anomaly_count": len(anomalous_orders),
49        "anomalous_order_ids": [o["order_id"] for o in anomalous_orders],
50        "mean_usd": round(mean_val, 2),
51        "std_usd": round(std_val, 2),
52        "high_risk_customer_count": sum(1 for score in customer_risk.values() if score > 3.0)
53    }
54"#;

123const AST_VALIDATOR_SCRIPT: &str = r#"
124import ast
125import json
126import sys
127
128payload = json.loads(sys.stdin.read())
129code = payload["code"]
130allowed_modules = set(payload["allowed_modules"])
131forbidden_calls = {
132    "open", "eval", "exec", "compile", "__import__",
133    "getattr", "setattr", "delattr", "globals", "locals",
134    "breakpoint", "input"
135}
136forbidden_attrs = {
137    "__subclasses__", "__bases__", "__globals__", "__builtins__",
138    "__class__", "__mro__", "__code__", "__closure__"
139}
140
141violations = []
142try:
143    tree = ast.parse(code)
144except SyntaxError as exc:
145    print(json.dumps({"violations": [f"SyntaxError: {exc.msg} at line {exc.lineno}"]}))
146    sys.exit(0)
147
148has_compute_fn = False
149for node in ast.walk(tree):
150    if isinstance(node, ast.FunctionDef) and node.name == "compute":
151        has_compute_fn = True
152    elif isinstance(node, ast.Import):
153        for alias in node.names:
154            root_mod = alias.name.split(".")[0]
155            if root_mod not in allowed_modules:
156                violations.append(f"Forbidden import '{alias.name}' (allowed: {sorted(allowed_modules)})")
157    elif isinstance(node, ast.ImportFrom):
158        mod_name = (node.module or "").split(".")[0]
159        if mod_name not in allowed_modules:
160            violations.append(f"Forbidden from-import '{node.module}' (allowed: {sorted(allowed_modules)})")
161    elif isinstance(node, ast.Call):
162        if isinstance(node.func, ast.Name) and node.func.id in forbidden_calls:
163            violations.append(f"Forbidden builtin call '{node.func.id}()'")
164    elif isinstance(node, ast.Attribute):
165        if node.attr in forbidden_attrs or node.attr.startswith("__"):
166            violations.append(f"Forbidden dunder attribute access '{node.attr}'")
167
168if not has_compute_fn:
169    violations.append("Missing required top-level entrypoint 'def compute(rows):'")
170
171print(json.dumps({"violations": violations}))
172"#;

225pub fn validate_python_algorithm(
226    req: &PythonAlgorithmValidationRequest,
227    config: &SandboxConfig,
228) -> Result<PythonAlgorithmValidationResult, SandboxError> {
229    let mut hasher = Sha256::new();
230    hasher.update(req.code.as_bytes());
231    let code_sha256 = format!("{:x}", hasher.finalize());
232
233    let mut violations = Vec::new();
234    if req.code.trim().is_empty() {
235        violations.push("Python algorithm code must not be empty".to_string());
236    }
237    if req.code.len() > config.max_code_bytes {
238        violations.push(format!(
239            "Python algorithm code length ({}) exceeds max_code_bytes ({})",
240            req.code.len(),
241            config.max_code_bytes
242        ));
243    }
244
245    if violations.is_empty() {
246        let input_json = serde_json::json!({
247            "code": req.code,
248            "allowed_modules": config.allowed_modules,
249        });
250        let mut child = Command::new(&config.python_binary)
251            .arg("-I")
252            .arg("-c")
253            .arg(AST_VALIDATOR_SCRIPT)
254            .env_clear()
255            .stdin(Stdio::piped())
256            .stdout(Stdio::piped())
257            .stderr(Stdio::piped())
258            .spawn()?;
259
260        if let Some(mut stdin) = child.stdin.take() {
261            stdin.write_all(input_json.to_string().as_bytes())?;
262        }
263        let output = child.wait_with_output()?;
264        if !output.status.success() {
265            let stderr = String::from_utf8_lossy(&output.stderr).to_string();
266            return Err(SandboxError::RuntimeFailure(stderr));
267        }
268        let parsed: serde_json::Value = serde_json::from_slice(&output.stdout)?;
269        if let Some(arr) = parsed.get("violations").and_then(|v| v.as_array()) {
270            for item in arr {
271                if let Some(msg) = item.as_str() {
272                    violations.push(msg.to_string());
273                }
274            }
275        }
276    }
277
278    let is_valid = violations.is_empty();
279    let aad_skill_manifest = if is_valid {
280        Some(AadSkillPromotionManifest {
281            skill_name: req.skill_name.clone(),
282            algorithm_version: "1.0.0".to_string(),
283            code_sha256: code_sha256.clone(),
284            description: req.description.clone(),
285            allowed_modules: config.allowed_modules.clone(),
286            mcp_tool_binding: "algorithm_execute".to_string(),
287        })
288    } else {
289        None
290    };
291
292    Ok(PythonAlgorithmValidationResult {
293        is_valid,
294        code_sha256,
295        violations,
296        aad_skill_manifest,
297    })
298}

300fn read_parquet_rows_as_json(
301    path: &Path,
302) -> Result<Vec<BTreeMap<String, serde_json::Value>>, SandboxError> {
303    let file = File::open(path)?;
304    let builder = ParquetRecordBatchReaderBuilder::try_new(file)?;
305    let reader = builder.build()?;
306    let mut rows = Vec::new();
307
308    for batch_res in reader {
309        let batch = batch_res?;
310        let schema = batch.schema();
311        let num_rows = batch.num_rows();
312        let mut batch_rows = vec![BTreeMap::new(); num_rows];
313
314        for (col_idx, field) in schema.fields().iter().enumerate() {
315            let col = batch.column(col_idx);
316            let name = field.name().clone();
317            match field.data_type() {
318                DataType::Utf8 => {
319                    if let Some(arr) = col.as_any().downcast_ref::<StringArray>() {
320                        for (r_idx, row_map) in batch_rows.iter_mut().enumerate() {
321                            if arr.is_null(r_idx) {
322                                row_map.insert(name.clone(), serde_json::Value::Null);
323                            } else {
324                                row_map.insert(
325                                    name.clone(),
326                                    serde_json::Value::String(arr.value(r_idx).to_string()),
327                                );
328                            }
329                        }
330                    }
331                }
332                DataType::Float64 => {
333                    if let Some(arr) = col.as_any().downcast_ref::<Float64Array>() {
334                        for (r_idx, row_map) in batch_rows.iter_mut().enumerate() {
335                            if arr.is_null(r_idx) {
336                                row_map.insert(name.clone(), serde_json::Value::Null);
337                            } else {
338                                row_map.insert(
339                                    name.clone(),
340                                    serde_json::json!(arr.value(r_idx)),
341                                );
342                            }
343                        }
344                    }
345                }
346                DataType::Int64 => {
347                    if let Some(arr) = col.as_any().downcast_ref::<Int64Array>() {
348                        for (r_idx, row_map) in batch_rows.iter_mut().enumerate() {
349                            if arr.is_null(r_idx) {
350                                row_map.insert(name.clone(), serde_json::Value::Null);
351                            } else {
352                                row_map.insert(
353                                    name.clone(),
354                                    serde_json::json!(arr.value(r_idx)),
355                                );
356                            }
357                        }
358                    }
359                }
360                _ => {}
361            }
362        }
363        rows.extend(batch_rows);
364    }
365
366    Ok(rows)
367}

369pub fn execute_python_algorithm(
370    req: &PythonAlgorithmExecuteRequest,
371    config: &SandboxConfig,
372    provenance_store: &ProvenanceStore,
373) -> Result<PythonAlgorithmExecuteResult, SandboxError> {
374    let validation = validate_python_algorithm(
375        &PythonAlgorithmValidationRequest {
376            skill_name: req.skill_name.clone(),
377            description: req.description.clone(),
378            code: req.code.clone(),
379        },
380        config,
381    )?;
382
383    if !validation.is_valid {
384        return Ok(PythonAlgorithmExecuteResult {
385            executed: false,
386            validation,
387            output: None,
388            provenance_run_id: None,
389        });
390    }
391
392    let rows = read_parquet_rows_as_json(Path::new(&req.parquet_path))?;
393    let payload = serde_json::json!({
394        "code": req.code,
395        "rows": rows,
396    });
397
398    let mut child = Command::new(&config.python_binary)
399        .arg("-I")
400        .arg("-c")
401        .arg(SANDBOX_RUNNER_SCRIPT)
402        .env_clear()
403        .stdin(Stdio::piped())
404        .stdout(Stdio::piped())
405        .stderr(Stdio::piped())
406        .spawn()?;
407
408    if let Some(mut stdin) = child.stdin.take() {
409        stdin.write_all(payload.to_string().as_bytes())?;
410    }
411
412    let start = Instant::now();
413    let timeout = Duration::from_millis(config.timeout_ms);
414    loop {
415        if let Some(_status) = child.try_wait()? {
416            break;
417        }
418        if start.elapsed() >= timeout {
419            let _ = child.kill();
420            return Err(SandboxError::Timeout(config.timeout_ms));
421        }
422        std::thread::sleep(Duration::from_millis(5));
423    }
424
425    let output = child.wait_with_output()?;
426    if !output.status.success() {
427        let stderr = String::from_utf8_lossy(&output.stderr).to_string();
428        return Err(SandboxError::RuntimeFailure(stderr));
429    }
430
431    let parsed_output: serde_json::Value = serde_json::from_slice(&output.stdout)?;
432    let anomaly_count = parsed_output
433        .get("anomaly_count")
434        .and_then(|v| v.as_f64())
435        .unwrap_or(0.0);
436
437    let epistemic_status = if req.verified_by.is_some() {
438        "verified"
439    } else {
440        "empirically_tested"
441    };
442    let mut verified_by_list = Vec::new();
443    if let Some(v) = &req.verified_by {
444        verified_by_list.push(v.clone());
445    }
446
447    let run = provenance_store.record_run(
448        &req.skill_name,
449        vec![req.dataset_name.clone()],
450        vec![format!(
451            "aad_skill:{}:{}:sha256:{}",
452            req.skill_name, epistemic_status, validation.code_sha256
453        )],
454        anomaly_count,
455        TrustSummary {
456            all_rules_human_verified: req.verified_by.is_some(),
457            min_rule_confidence: if req.verified_by.is_some() { 1.0 } else { 0.85 },
458            verified_by_stewards: verified_by_list,
459            pre_aggregation_applied: false,
460            unit_conversion_applied: false,
461        },
462    )?;
463
464    Ok(PythonAlgorithmExecuteResult {
465        executed: true,
466        validation,
467        output: Some(parsed_output),
468        provenance_run_id: Some(run.provenance_run_id),
469    })
470}
